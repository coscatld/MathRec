## 一、问题描述与数学模型

### 1. 已知条件
| 参数 | 含义 |
|:---|:---|
| 区域 | 边长为 $L$ 的正方形（如 $160 \times 160$ km） |
| $n$ 架飞机 | 每架有位置 $(x_i, y_i)$、初始方向角 $\theta_i$（弧度）、速度 $v$（相同） |
| 安全距离 $d$ | 任意两架飞机距离必须始终 $> d$（如 8 km） |
| 最大调整角度 $\Delta\theta_{\max}$ | 每架飞机方向角调整量不能超过此值（如 $30^\circ$） |

### 2. 飞行轨迹
第 $i$ 架飞机在时刻 $t$ 的位置：
$$
\begin{cases}
x_i(t) = x_i + v \cdot t \cdot \cos(\theta_i + \Delta\theta_i) \\
y_i(t) = y_i + v \cdot t \cdot \sin(\theta_i + \Delta\theta_i)
\end{cases}
$$

### 3. 碰撞约束
任意两架飞机 $i, j$ 在区域内飞行期间，距离的平方始终大于 $d^2$：

$$
\min_{t \geq 0} \left[ (x_i(t)-x_j(t))^2 + (y_i(t)-y_j(t))^2 \right] > d^2
$$

### 4. 目标函数（常见三种）

| 目标       | 表达式                                      | 含义             |     |              |
| :------- | :--------------------------------------- | :------------- | --- | ------------ |
| 总调整角度最小  | $\min \sum_{i=1}^{n}$                    | \Delta\theta_i | $   | 所有飞机调整量之和最小  |
| 最大调整角度最小 | $\min \max_{i}$                          | \Delta\theta_i | $   | 最"倒霉"的飞机调整最少 |
| 总调整量平方最小 | $\min \sum_{i=1}^{n} (\Delta\theta_i)^2$ | 平滑惩罚，便于求导      |     |              |

## 二、碰撞约束的简化推导

两架飞机 $i, j$ 的相对位置：
$$
\begin{cases}
\Delta x_{ij} = x_i - x_j \\
\Delta y_{ij} = y_i - y_j \\
\Delta v_x = v[\cos(\theta_i+\Delta\theta_i) - \cos(\theta_j+\Delta\theta_j)] \\
\Delta v_y = v[\sin(\theta_i+\Delta\theta_i) - \sin(\theta_j+\Delta\theta_j)]
\end{cases}
$$

相对距离的平方：
$$
D_{ij}^2(t) = (\Delta x_{ij} + \Delta v_x \cdot t)^2 + (\Delta y_{ij} + \Delta v_y \cdot t)^2
$$

这是关于 $t$ 的二次函数，最小值出现在：
$$
t^* = -\frac{\Delta x_{ij} \cdot \Delta v_x + \Delta y_{ij} \cdot \Delta v_y}{\Delta v_x^2 + \Delta v_y^2}
$$

- 若 $t^* < 0$，则最小距离在 $t=0$ 处取得（初始时刻）
- 若 $t^* \geq 0$，最小距离在 $t^*$ 处取得

因此碰撞约束可写为：
$$
D_{ij}^2(t^*) \geq d^2 \quad \text{（若 } t^* \geq 0 \text{）}
$$

## 三、MATLAB 完整求解代码

```matlab
%% 飞行管理问题 - MATLAB求解
% 使用 fmincon 求解最小总调整角度问题

clear; clc; close all;

%% ========== 参数设置 ==========
L = 160;           % 区域边长 (km)
v = 800;           % 飞机速度 (km/h)
d = 8;             % 安全距离 (km)
n = 5;             % 飞机数量
dtheta_max = pi/6; % 最大调整角度 (30度)

% 飞机初始数据 [x, y, theta(弧度)]
% 示例数据（请替换为你的实际数据）
planes = [
    150, 140, 243*pi/180;   % 飞机1
    85,  85,  236*pi/180;   % 飞机2
    150, 155, 220*pi/180;   % 飞机3
    145, 50,  159*pi/180;   % 飞机4
    130, 150, 230*pi/180;   % 飞机5
];

x0 = planes(:,1);
y0 = planes(:,2);
theta0 = planes(:,3);

%% ========== 优化设置 ==========
% 决策变量：各飞机的角度调整量 dtheta_i
% 初始猜测：不调整
dtheta0 = zeros(n, 1);

% 上下界约束
lb = -dtheta_max * ones(n, 1);
ub = dtheta_max * ones(n, 1);

% 目标函数：最小化总调整量（可用绝对值或平方）
% 方法1：最小二乘（光滑，易求导）
obj_fun = @(dtheta) sum(dtheta.^2);

% 方法2：最小化最大调整量（minimax，需改写）
% obj_fun = @(dtheta) max(abs(dtheta));

% 非线性约束函数
nonlcon = @(dtheta) collision_constraints(dtheta, x0, y0, theta0, v, d);

% 优化选项
options = optimoptions('fmincon', ...
    'Display', 'iter', ...
    'Algorithm', 'sqp', ...
    'MaxIterations', 1000, ...
    'MaxFunctionEvaluations', 5000, ...
    'TolFun', 1e-8, ...
    'TolCon', 1e-6);

%% ========== 求解 ==========
[dtheta_opt, fval, exitflag] = fmincon(obj_fun, dtheta0, [], [], [], [], lb, ub, nonlcon, options);

% 输出结果
fprintf('\n========== 求解结果 ==========\n');
for i = 1:n
    fprintf('飞机 %d: 调整角度 = %.4f° (%.4f rad)\n', i, dtheta_opt(i)*180/pi, dtheta_opt(i));
end
fprintf('目标函数值 = %.6f\n', fval);
fprintf('退出标志 = %d\n', exitflag);

%% ========== 可视化 ==========
figure('Position', [100 100 800 800]);
hold on; grid on; axis equal;
xlim([0 L]); ylim([0 L]);
xlabel('x (km)'); ylabel('y (km)');
title('飞行管理问题 - 最优调整方案');

% 绘制区域边界
rectangle('Position', [0 0 L L], 'EdgeColor', 'k', 'LineWidth', 1.5);

% 绘制飞机初始位置和调整后的飞行轨迹
colors = lines(n);
t_max = L/v * 2;  % 模拟时间

for i = 1:n
    % 初始方向
    dx0 = cos(theta0(i));
    dy0 = sin(theta0(i));
    
    % 调整后方向
    theta_new = theta0(i) + dtheta_opt(i);
    dx = cos(theta_new);
    dy = sin(theta_new);
    
    % 绘制初始位置
    plot(x0(i), y0(i), 'o', 'Color', colors(i,:), 'MarkerSize', 10, 'LineWidth', 2);
    
    % 绘制初始方向箭头
    quiver(x0(i), y0(i), dx0*10, dy0*10, 0, 'Color', colors(i,:), ...
        'LineStyle', '--', 'MaxHeadSize', 0.5, 'LineWidth', 1);
    
    % 绘制调整后方向箭头
    quiver(x0(i), y0(i), dx*15, dy*15, 0, 'Color', colors(i,:), ...
        'LineWidth', 2.5);
    
    % 绘制飞行轨迹
    t = linspace(0, t_max, 100);
    x_traj = x0(i) + v*t*cos(theta_new);
    y_traj = y0(i) + v*t*sin(theta_new);
    
    % 只绘制在区域内的部分
    in_region = (x_traj >= 0) & (x_traj <= L) & (y_traj >= 0) & (y_traj <= L);
    plot(x_traj(in_region), y_traj(in_region), '-', 'Color', colors(i,:), 'LineWidth', 1.5);
    
    % 标注
    text(x0(i)+3, y0(i)+3, sprintf('飞机%d', i), 'Color', colors(i,:), 'FontSize', 11);
end

% 绘制安全距离圆（在初始位置处示意）
for i = 1:n
    for j = i+1:n
        % 计算最小距离
        [~, ~, min_dist] = compute_min_distance(dtheta_opt, x0, y0, theta0, v, i, j);
        fprintf('飞机 %d 与 %d 的最小距离 = %.4f km\n', i, j, min_dist);
    end
end

legend('区域边界', 'Location', 'best');
hold off;

%% ========== 非线性约束函数 ==========
function [c, ceq] = collision_constraints(dtheta, x0, y0, theta0, v, d)
    n = length(dtheta);
    ceq = [];  % 没有等式约束
    
    % 不等式约束：c <= 0
    % 要求最小距离的平方 >= d^2，即 d^2 - min_dist^2 <= 0 的反面
    % 实际上：min_dist^2 - d^2 >= 0，写成 -min_dist^2 + d^2 <= 0
    num_pairs = n*(n-1)/2;
    c = zeros(num_pairs, 1);
    
    idx = 1;
    for i = 1:n
        for j = i+1:n
            [~, ~, min_dist] = compute_min_distance(dtheta, x0, y0, theta0, v, i, j);
            % 约束：最小距离必须 >= d，即 d - min_dist <= 0
            c(idx) = d - min_dist;
            idx = idx + 1;
        end
    end
end

%% ========== 计算两架飞机的最小距离 ==========
function [t_star, dist_sq_min, min_dist] = compute_min_distance(dtheta, x0, y0, theta0, v, i, j)
    theta_i = theta0(i) + dtheta(i);
    theta_j = theta0(j) + dtheta(j);
    
    dx = x0(i) - x0(j);
    dy = y0(i) - y0(j);
    
    dvx = v*(cos(theta_i) - cos(theta_j));
    dvy = v*(sin(theta_i) - sin(theta_j));
    
    a = dvx^2 + dvy^2;
    b = 2*(dx*dvx + dy*dvy);
    
    if a < 1e-10  % 两架飞机速度几乎相同（平行飞行）
        t_star = 0;
        dist_sq_min = dx^2 + dy^2;
    else
        t_star = -b / (2*a);
        if t_star < 0
            t_star = 0;
        end
        dist_sq_min = (dx + dvx*t_star)^2 + (dy + dvy*t_star)^2;
    end
    
    min_dist = sqrt(dist_sq_min);
end
```
